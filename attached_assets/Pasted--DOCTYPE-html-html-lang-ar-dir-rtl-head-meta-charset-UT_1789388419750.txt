    <script>
        let qrGenerator = null;

        function handleLogin(e) {
            e.preventDefault();
            const role = document.getElementById('user-role').value;
            document.getElementById('logged-user-role').innerText = role;
            document.getElementById('login-modal').classList.add('hidden');
        }

        function logout() {
            document.getElementById('login-modal').classList.remove('hidden');
        }

        function switchView(viewName) {
            document.querySelectorAll('main > section').forEach(el => el.classList.add('hidden'));
            document.getElementById(`view-${viewName}`).classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.classList.remove('active-tab');
                btn.classList.add('text-zinc-400');
            });
            const activeNav = document.getElementById(`nav-${viewName}`);
            activeNav.classList.add('active-tab');
            activeNav.classList.remove('text-zinc-400');
        }

        function generateQRPreview(url) {
            const container = document.getElementById('qr-preview-container');
            const qrDiv = document.getElementById('qrcode');
            if(!url) {
                container.classList.add('hidden');
                return;
            }
            container.classList.remove('hidden');
            qrDiv.innerHTML = "";
            new QRCode(qrDiv, {
                text: url,
                width: 90,
                height: 90
            });
        }

        function generateReport(e) {
            e.preventDefault();
            alert("تم جمع البيانات وتوليد التقرير بنجاح! جاهز للتنزيل المباشر بصيغة PDF.");
        }
